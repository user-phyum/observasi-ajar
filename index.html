Kode Google Apps Script Backend

Gunakan kode di bawah ini pada Google Sheets Anda untuk menyimpan otomatis isian form.

Langkah Integrasi Ke Google Sheet:

  1. Buka Google Spreadsheet tujuan Anda.
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Hapus semua kode default dan tempelkan (paste) kode di bawah ini.
  4. Klik Deploy (Terapkan) > New Deployment (Penerapan Baru).
  5. Pilih tipe Web App, atur Who has access ke Anyone (Siapa Saja), lalu klik Deploy.
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Observasi') || ss.getActiveSheet();
    
    // Buat Header jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'Timestamp',
        'Email Pengamat',
        'Tanggal & Jam Perkuliahan',
        'Nama Mata Kuliah',
        'Nama Offering',
        '1. Sesuai Jadwal',
        '2. Dosen Tepat Waktu (<15m)',
        '3. Sesuai RPS',
        'URL Foto / Dokumen',
        'Kendala / Catatan'
      ]);
      sheet.getRange(1, 1, 1, 10).setFontWeight('bold').setBackground('#0284c7').setFontColor('#ffffff');
    }
    
    var data = JSON.parse(e.postData.contents);
    
    // Tambah baris data baru
    sheet.appendRow([
      data.timestamp || new Date().toLocaleString("en-US", {timeZone: "Asia/Jakarta"}),
      data.email || '',
      data.tanggalJam || '',
      data.matakuliah || '',
      data.offering || '',
      data.q1 || '',
      data.q2 || '',
      data.q3 || '',
      data.foto || '',
      data.kendala || ''
    ]);
    
    return ContentService
      .createTextOutput(JSON.stringify({ "status": "success", "message": "Data berhasil disimpan" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
